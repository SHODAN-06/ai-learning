### 1. Простой калькулятор в консоли

Этот вариант использует функции для каждой операции и цикл `while`, чтобы программа не закрывалась после одного вычисления.

```python
# calculator.py

def add(x, y):
    """Сложение"""
    return x + y

def subtract(x, y):
    """Вычитание"""
    return x - y

def multiply(x, y):
    """Умножение"""
    return x * y

def divide(x, y):
    """Деление с обработкой деления на ноль"""
    if y == 0:
        return "Ошибка! Деление на ноль."
    return x / y

def calculator():
    """Основная функция калькулятора"""
    print("Выберите операцию:")
    print("1. Сложение")
    print("2. Вычитание")
    print("3. Умножение")
    print("4. Деление")

    while True:
        # Проверка, что пользователь ввел число от 1 до 4
        choice = input("Введите номер операции (1/2/3/4): ")

        if choice in ('1', '2', '3', '4'):
            try:
                num1 = float(input("Введите первое число: "))
                num2 = float(input("Введите второе число: "))
            except ValueError:
                print("Неверный ввод. Пожалуйста, введите числа.")
                continue

            if choice == '1':
                print(f"{num1} + {num2} = {add(num1, num2)}")

            elif choice == '2':
                print(f"{num1} - {num2} = {subtract(num1, num2)}")

            elif choice == '3':
                print(f"{num1} * {num2} = {multiply(num1, num2)}")

            elif choice == '4':
                result = divide(num1, num2)
                print(f"{num1} / {num2} = {result}")

        else:
            print("Неверный ввод. Пожалуйста, выберите существующий вариант.")

        # Спрашиваем, хочет ли пользователь продолжить
        next_calculation = input("Хотите выполнить еще одну операцию? (да/нет): ")
        if next_calculation.lower() != 'да':
            print("Калькулятор закрыт.")
            break

if __name__ == "__main__":
    calculator()
```

**Как запустить:**
1.  Сохраните этот код в файл с именем `calculator.py`.
2.  Откройте терминал или командную строку.
3.  Перейдите в папку с файлом.
4.  Выполните команду: `python calculator.py`

### 2. Калькулятор с графическим интерфейсом (GUI) с помощью Tkinter

Этот вариант создает полноценное окно с кнопками, как у настоящего калькулятора. Tkinter входит в стандартную библиотеку Python, поэтому ничего дополнительно устанавливать не нужно.

```python
# gui_calculator.py

import tkinter as tk

def click_button(value):
    """Обработка нажатия на кнопки с цифрами и операторами"""
    current = entry.get()
    entry.delete(0, tk.END)
    entry.insert(0, current + str(value))

def clear_field():
    """Очистка поля ввода"""
    entry.delete(0, tk.END)

def calculate():
    """Вычисление результата"""
    try:
        result = eval(entry.get())
        entry.delete(0, tk.END)
        entry.insert(0, str(result))
    except Exception as e:
        entry.delete(0, tk.END)
        entry.insert(0, "Ошибка")

# Создание главного окна
root = tk.Tk()
root.title("Калькулятор")
root.geometry("312x324")
root.resizable(0, 0)

# Поле ввода
entry = tk.Entry(root, font=('arial', 18, 'bold'), bd=10, insertwidth=4, width=14, justify='right')
entry.grid(row=0, column=0, columnspan=4)

# Определение кнопок
button_texts = [
    ('7', 1, 0), ('8', 1, 1), ('9', 1, 2), ('/', 1, 3),
    ('4', 2, 0), ('5', 2, 1), ('6', 2, 2), ('*', 2, 3),
    ('1', 3, 0), ('2', 3, 1), ('3', 3, 2), ('-', 3, 3),
    ('C', 4, 0), ('0', 4, 1), ('.', 4, 2), ('+', 4, 3),
    ('=', 5, 0, 4) # Кнопка "=" занимает 4 колонки
]

# Создание и размещение кнопок
for (text, row, col, *span) in button_texts:
    colspan = span[0] if span else 1
    
    if text == '=':
        btn = tk.Button(root, text=text, height=3, width=13, font=('arial', 18, 'bold'),
                        command=calculate)
    elif text == 'C':
        btn = tk.Button(root, text=text, height=3, width=13, font=('arial', 18, 'bold'),
                        command=clear_field)
    else:
        btn = tk.Button(root, text=text, height=3, width=13, font=('arial', 18, 'bold'),
                        command=lambda t=text: click_button(t))
    
    btn.grid(row=row, column=col, columnspan=colspan, padx=1, pady=1)

# Запуск главного цикла окна
root.mainloop()
```

**Как запустить:**
1.  Сохраните код в файл `gui_calculator.py`.
2.  Запустите его из терминала: `python gui_calculator.py`.
3.  Откроется окно с вашим калькулятором.

### 3. Калькулятор-выражений (как в командной строке)

Этот вариант самый короткий. Он позволяет вводить сразу целое математическое выражение, например, `(10 + 5) * 3 - 2`.

> **Важно:** Использование `eval()` может быть небезопасным, если вы обрабатываете ввод от неизвестных пользователей, так как он выполнит любой код. Для личного использования это самый быстрый способ.

```python
# expression_calculator.py

def simple_calc():
    """Простой калькулятор для вычисления выражений"""
    print("Введите математическое выражение (например, 2 + 2 * 3)")
    print("Для выхода введите 'выход' или 'exit'")
    
    while True:
        user_input = input(">>> ")
        
        if user_input.lower() in ['выход', 'exit', 'quit']:
            print("До свидания!")
            break
            
        try:
            # eval вычисляет строку как выражение Python
            result = eval(user_input)
            print(f"Результат: {result}")
        except Exception as e:
            print(f"Ошибка в выражении: {e}")

if __name__ == "__main__":
    simple_calc()
```
